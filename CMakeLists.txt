cmake_minimum_required(VERSION 3.24)
project(Cere VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Quick Qml Network Widgets DBus)
find_package(LayerShellQt REQUIRED)
set(CERE_NATIVE native/controller.cpp native/controller.h native/motion.cpp native/motion.h native/avatarpuppet.cpp native/avatarpuppet.h native/avatarskeleton.h)
qt_add_executable(cere native/main.cpp ${CERE_NATIVE})
target_link_libraries(cere PRIVATE Qt6::Core Qt6::Gui Qt6::Quick Qt6::Qml Qt6::Network Qt6::Widgets Qt6::DBus LayerShellQt::Interface)
target_compile_definitions(cere PRIVATE CERE_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(cere PROPERTIES INSTALL_RPATH "$ORIGIN")
include(GNUInstallDirs)
find_program(NODE_EXECUTABLE node REQUIRED)
set(CERE_ICON_SIZES 16 22 24 32 48 64 128 256 512)
set(CERE_ICON_OUTPUTS)
foreach(CERE_ICON_SIZE IN LISTS CERE_ICON_SIZES)
  list(APPEND CERE_ICON_OUTPUTS "${CMAKE_CURRENT_BINARY_DIR}/icons/hicolor/${CERE_ICON_SIZE}x${CERE_ICON_SIZE}/apps/cere.png")
endforeach()
add_custom_command(
  OUTPUT ${CERE_ICON_OUTPUTS}
  COMMAND ${NODE_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/prepare-icons.mjs
    ${CMAKE_CURRENT_SOURCE_DIR}/assets/cere-emblem.png ${CMAKE_CURRENT_BINARY_DIR}/icons/hicolor
  DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/assets/cere-emblem.png ${CMAKE_CURRENT_SOURCE_DIR}/tools/prepare-icons.mjs
  COMMENT "Preparing Cere application icons"
  VERBATIM)
add_custom_target(cere-icons ALL DEPENDS ${CERE_ICON_OUTPUTS})
add_dependencies(cere cere-icons)
find_package(Python3 3.12 REQUIRED COMPONENTS Interpreter)
find_path(NODE_INCLUDE_DIR node_api.h PATH_SUFFIXES node REQUIRED)
add_library(cere-peercred MODULE native/peercred.cpp)
target_include_directories(cere-peercred PRIVATE ${NODE_INCLUDE_DIR})
set_target_properties(cere-peercred PROPERTIES PREFIX "" SUFFIX ".node" AUTOMOC OFF)
execute_process(COMMAND ${NODE_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/stage-runtime.mjs ${CMAKE_CURRENT_BINARY_DIR}/runtime RESULT_VARIABLE CERE_RUNTIME_RESULT)
if(NOT CERE_RUNTIME_RESULT EQUAL 0)
  message(FATAL_ERROR "Could not stage Cere runtime dependencies; run npm ci first")
endif()
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/prepare-tts.py
  --output ${CMAKE_CURRENT_BINARY_DIR}/tts --arch ${CMAKE_SYSTEM_PROCESSOR}
  RESULT_VARIABLE CERE_TTS_RESULT)
if(NOT CERE_TTS_RESULT EQUAL 0)
  message(FATAL_ERROR "Could not stage offline Piper speech assets; see tools/prepare-tts.py")
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  qt_add_executable(cere-ui-check tests/ui.cpp ${CERE_NATIVE})
  target_link_libraries(cere-ui-check PRIVATE Qt6::Core Qt6::Gui Qt6::Quick Qt6::Qml Qt6::Network Qt6::Widgets Qt6::DBus Qt6::Test LayerShellQt::Interface)
  target_compile_definitions(cere-ui-check PRIVATE CERE_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  qt_add_executable(cere-motion-check tests/motion.cpp native/motion.cpp native/motion.h)
  target_link_libraries(cere-motion-check PRIVATE Qt6::Core Qt6::Gui Qt6::Test)
  # Checked standard-library preconditions (for example std::clamp's ordered bounds).
  target_compile_definitions(cere-motion-check PRIVATE CERE_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}" _GLIBCXX_ASSERTIONS)
  add_test(NAME motion COMMAND cere-motion-check)
  qt_add_executable(cere-avatar-check tests/avatarpuppet.cpp native/avatarpuppet.cpp native/avatarpuppet.h native/avatarskeleton.h)
  target_include_directories(cere-avatar-check PRIVATE native)
  target_link_libraries(cere-avatar-check PRIVATE Qt6::Core Qt6::Gui Qt6::Quick Qt6::Qml)
  add_test(NAME avatar COMMAND cere-avatar-check ${CMAKE_CURRENT_SOURCE_DIR})
  set_tests_properties(avatar PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=rhi;QSG_RHI_BACKEND=opengl")
  add_test(NAME avatar-software COMMAND cere-avatar-check ${CMAKE_CURRENT_SOURCE_DIR})
  set_tests_properties(avatar-software PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;CERE_PUPPET_TEST_SOFTWARE=1")
endif()
install(TARGETS cere RUNTIME DESTINATION ${CMAKE_INSTALL_LIBDIR}/cere)
install(PROGRAMS tools/cere DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY qml assets broker DESTINATION ${CMAKE_INSTALL_DATADIR}/cere)
install(DIRECTORY protocol DESTINATION ${CMAKE_INSTALL_DATADIR}/cere)
install(DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/tts DESTINATION ${CMAKE_INSTALL_DATADIR}/cere USE_SOURCE_PERMISSIONS)
install(FILES tools/setup-speech.ts DESTINATION ${CMAKE_INSTALL_DATADIR}/cere/tools)
install(TARGETS cere-peercred LIBRARY DESTINATION ${CMAKE_INSTALL_DATADIR}/cere/native)
install(DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/runtime/node_modules DESTINATION ${CMAKE_INSTALL_DATADIR}/cere)
install(FILES package.json package-lock.json DESTINATION ${CMAKE_INSTALL_DATADIR}/cere)
install(FILES packaging/cere.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
foreach(CERE_ICON_SIZE IN LISTS CERE_ICON_SIZES)
  install(FILES ${CMAKE_CURRENT_BINARY_DIR}/icons/hicolor/${CERE_ICON_SIZE}x${CERE_ICON_SIZE}/apps/cere.png
    DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/${CERE_ICON_SIZE}x${CERE_ICON_SIZE}/apps)
endforeach()
install(FILES packaging/cere-broker.service DESTINATION ${CMAKE_INSTALL_LIBDIR}/systemd/user)
install(FILES README.md VALIDATION.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/cere)
install(FILES docs/tts.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/cere/docs)
install(FILES packaging/tts-NOTICES.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/cere/packaging)
install(DIRECTORY docs/memory DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/cere)
install(DIRECTORY docs/mobile DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/cere)
install(FILES packaging/memory-compose.yml packaging/memory.env.example DESTINATION ${CMAKE_INSTALL_DATADIR}/cere/packaging)
install(PROGRAMS tools/memory-dependencies.sh DESTINATION ${CMAKE_INSTALL_DATADIR}/cere/tools)
install(FILES tools/check-memory-adapters.ts DESTINATION ${CMAKE_INSTALL_DATADIR}/cere/tools)
